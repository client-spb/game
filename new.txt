Раздел 7. Звук
Все звуки генерируй через Web Audio API на JavaScript.
Нельзя подключать внешние аудиофайлы, если я явно не разрешил это.
Звуки интерфейса и игровых действий обязательны.
Желательно делать целые короткие мелодии для важных событий: победа, поражение, финал, новый уровень, получение бонуса, ошибка.
Фоновая музыка делается только если она подходит жанру и не вредит производительности.
В игре всегда должна быть кнопка выключения звука.
Настройка звука должна сохраняться.
Первый запуск звука должен корректно разблокироваться после первого пользовательского касания.
Примерная структура звукового движка:
const FX = (() => {
  let ctx = null;
  function audio() {
    if (!ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) return null;
      ctx = new AC();
    }
    if (ctx.state === 'suspended') ctx.resume();
    return ctx;
  }
  function tone(freq, dur, type, vol, delay, slide) {
    const ac = audio();
    if (!ac || !SAVE.sound) return;
    const t = ac.currentTime + (delay || 0);
    const osc = ac.createOscillator();
    const gain = ac.createGain();
    osc.type = type || 'sine';
    osc.frequency.setValueAtTime(freq, t);
    if (slide) osc.frequency.exponentialRampToValueAtTime(slide, t + dur);
    gain.gain.setValueAtTime(vol, t);
    gain.gain.exponentialRampToValueAtTime(0.0001, t + dur);
    osc.connect(gain); gain.connect(ac.destination);
    osc.start(t); osc.stop(t + dur + 0.05);
  }
  return {
    unlock() { audio(); },
    click() { tone(800, 0.05, 'sine', 0.06); },
    win() { [523, 659, 784, 1047].forEach((f, i) => tone(f, 0.14, 'triangle', 0.14, i * 0.11)); },
    lose() { [330, 262, 208, 165].forEach((f, i) => tone(f, 0.2, 'sawtooth', 0.1, i * 0.16)); }
  };
})();
Можно расширять этот движок под жанр игры.
Раздел 8. Сохранения
Сохранения делай через обычный localStorage.
Запрещено сохранять прогресс каждый кадр.
Запрещено делать сотни записей в секунду.
Запрещено забивать localStorage мусором.
Используй один основной ключ на игру.
Ключ должен быть понятным и версионируемым, например:
const SAVE_KEY = 'game_slug_save_v1';
Сохраняй только по важным событиям: пройден уровень, проигран уровень, изменены настройки, открыт новый уровень, игра завершена, сброшен прогресс.
Если нужно автосохранение, используй ограничение частоты (debounce или throttle).
Все сохранения должны быть защищены через try catch.
function loadSave() {
  try {
    const raw = localStorage.getItem(SAVE_KEY);
    if (raw) Object.assign(SAVE, JSON.parse(raw));
  } catch (e) {}
}
function persist() {
  try {
    localStorage.setItem(SAVE_KEY, JSON.stringify(SAVE));
  } catch (e) {}
}

10.4. Отрисовка лидерборда
Лидерборд рисуется как список строк. Каждая строка содержит:
Медаль или номер места (1, 2, 3 — медали; остальные — номер)
Аватарку (если есть photo_url, иначе первая буква имени)
Имя игрока (name или nickname)
Счёт
Пример функции отрисовки:
function renderLeaderboard(players) {
  const list = $('lb-list');
  list.innerHTML = '';
  if (!players || players.length === 0) {
    list.innerHTML = '<div class="lb-empty">Пока нет рекордов. Будьте первым!</div>';
    return;
  }
  players.forEach((p, i) => {
    const row = document.createElement('div');
    row.className = 'lb-row';
    if (i === 0) row.classList.add('top-1');
    else if (i === 1) row.classList.add('top-2');
    else if (i === 2) row.classList.add('top-3');

    // Медаль
    const medal = document.createElement('div');
    medal.className = 'lb-medal';
    if (i === 0) { medal.textContent = '1'; medal.classList.add('gold'); }
    else if (i === 1) { medal.textContent = '2'; medal.classList.add('silver'); }
    else if (i === 2) { medal.textContent = '3'; medal.classList.add('bronze'); }
    else medal.textContent = '#' + (i + 1);
    row.appendChild(medal);

    // Аватарка
    const avatar = document.createElement('div');
    avatar.className = 'lb-avatar';
    const displayName = p.name || p.nickname || 'Игрок';
    const initial = displayName.charAt(0).toUpperCase();
    if (p.photo_url) {
      const img = document.createElement('img');
      img.src = p.photo_url;
      img.alt = '';
      img.onerror = function() { this.remove(); avatar.textContent = initial; };
      avatar.appendChild(img);
      avatar.appendChild(document.createTextNode(initial));
    } else {
      avatar.textContent = initial;
    }
    row.appendChild(avatar);

    // Имя
    const name = document.createElement('div');
    name.className = 'lb-name';
    name.textContent = displayName;
    row.appendChild(name);

    // Счёт
    const score = document.createElement('div');
    score.className = 'lb-score';
    score.textContent = p.score;
    row.appendChild(score);

    list.appendChild(row);
  });
}

Рекомендуемая логика шаринга текстом:
function shareText(text) {
  const link = 'https://max.ru/channel_igroteka_max';
  const fullText = text.includes(link)
    ? text
    : text + '\nиграй прямо в Максе без скачиваний ' + link;
  try {
    if (window.WebApp && window.WebApp.shareContent) {
      window.WebApp.shareContent({ text: fullText });
      return;
    }
  } catch (e) {}
  try {
    if (navigator.share) {
      navigator.share({ text: fullText });
      return;
    }
  } catch (e) {}
  // fallback: копирование в буфер
  try {
    if (navigator.clipboard) {
      navigator.clipboard.writeText(fullText);
      toast('Ссылка скопирована');
    }
  } catch (e) {}
}
canvas.toBlob(async blob => {
  try {
    if (!blob) return shareText(text);
    const file = new File([blob], 'share.png', { type: 'image/png' });
    if (navigator.canShare && navigator.canShare({ files: [file] })) {
      await navigator.share({ files: [file], text: fullText });
      return;
    }
  } catch (e) {}
  shareText(text);
}, 'image/png');

const MUS = { on: false, step: 0, next: 0, timer: 0, bpm: 112,
  lead: [329.6, 0, 392, 0, 440, 0, 392, 329.6, 293.7, 0, 329.6, 0, 392, 0, 493.9, 0],

function musicTick() {
  if (!MUS.on) return;
  const ac = FX.ctx(); if (!ac) return;
  try {
    while (MUS.next < ac.currentTime + 0.25) {
      const s = MUS.step % 16;
      const d = Math.max(0, MUS.next - ac.currentTime);
      if (MUS.lead[s]) FX.tone(MUS.lead[s], 0.15, 'triangle', 0.032, d);
      if (MUS.bass[s]) FX.tone(MUS.bass[s], 0.2, 'sine', 0.06, d);
      if (s % 2 === 0) FX.tone(2400, 0.02, 'square', 0.006, d);
      MUS.step++; MUS.next += 60 / MUS.bpm / 2;
    }
  } catch (e) {}
}
function startMusic() {
  stopMusic(); const ac = FX.ctx(); if (!ac) return;
  MUS.on = true; MUS.step = 0; MUS.next = ac.currentTime + 0.1;
  MUS.timer = setInterval(musicTick, 90);
}
function stopMusic() { MUS.on = false; if (MUS.timer) { clearInterval(MUS.timer); MUS.timer = 0; } }

function shareFallbackCopy(text) {
  try {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(() => toast('Ссылка скопирована')).catch(() => {}); return;

  const full = fullShareText(msg);
  try { if (window.WebApp && window.WebApp.shareContent) { window.WebApp.shareContent({ text: full }); return true; } } catch (e) {}
  try { if (navigator.share) { navigator.share({ text: full }).catch(() => {}); return true; } } catch (e) {}
  shareFallbackCopy(full); return true;
}

function rr(c, x, y, w, h, r) {
  r = Math.min(r, w / 2, h / 2);
  c.beginPath();
  c.moveTo(x + r, y);
  c.arcTo(x + w, y, x + w, y + h, r);
  c.arcTo(x + w, y + h, x, y + h, r);
  c.arcTo(x, y + h, x, y, r);
  c.arcTo(x, y, x + w, y, r);
  c.closePath();
}
function eyes(c, x, y, r, lookX, lookY, angry) {
  c.fillStyle = '#fff'; c.strokeStyle = INK; c.lineWidth = 2.5;
  for (const s of [-1, 1]) { c.beginPath(); c.arc(x + s * r * 1.15, y, r, 0, TAU); c.fill(); c.stroke(); }
  let dx = 0, dy = 0;
  const d = Math.hypot(lookX - x, lookY - y) || 1;
  dx = (lookX - x) / d * r * 0.38; dy = (lookY - y) / d * r * 0.38;
  c.fillStyle = INK;
  for (const s of [-1, 1]) { c.beginPath(); c.arc(x + s * r * 1.15 + dx, y + dy, r * 0.42, 0, TAU); c.fill(); }
  if (angry) {
    c.strokeStyle = INK; c.lineWidth = 3; c.lineCap = 'round';
    for (const s of [-1, 1]) {
      c.beginPath();
      c.moveTo(x + s * r * 1.9, y - r * 1.2);
      c.lineTo(x + s * r * 0.35, y - r * 0.55);
      c.stroke();
    }
  }
}

function levelKinds(n) {
  const k = ['crab'];
  if (n >= 2) k.push('diver');
  if (n >= 3) k.push('jelly');
  if (n >= 4) k.push('tank');
  if (n >= 6) k.push('mine');
  if (n >= 8) k.push('asteroid');
  if (n >= 12) k.push('sniper');
  if (n >= 16) k.push('carrier');
  if (n >= 20) k.push('ghost');
  if (n >= 25) k.push('laser_turret');
  if (n >= 30) k.push('swarm');
  if (n >= 40) k.push('mothership');
  if (n >= 50) k.push('elite');
  if (n >= 60) k.push('phantom');
  if (n >= 75) k.push('dreadnought');
  if (n >= 90) k.push('overlord');
  return k;
}

function spawnEnemy(kind, x, y, o) {
  o = o || {};
  const s = E_STATS[kind];
  const n = G.level;
  let hp = s.hp;
  if (kind === 'jelly' && n >= 8) hp = 3;
  if (kind === 'tank') hp = 5 + Math.floor(n / 3);
  if (kind === 'dreadnought') hp = 15 + Math.floor(n / 5);
  if (kind === 'mothership') hp = 12 + Math.floor(n / 4);
  if (kind === 'carrier') hp = 8 + Math.floor(n / 6);
  const elite = n >= 50 && kind === 'crab' && Math.random() < 0.15;
  if (elite) hp = Math.ceil(hp * 1.8);
  G.enemies.push({
    kind, x, y, baseX: x,
    vx: o.vx || 0, vy: o.vy || 70,
    amp: o.amp !== undefined ? o.amp : rand(28, 66),
    t: 0, hp, maxHp: hp, hitT: 0, r: s.r, pts: elite ? Math.ceil(s.pts * 1.5) : s.pts, drop: s.drop,
    fireT: rand(1.6, 3.2), state: 0, aimT: kind === 'diver' ? 0.85 : 0,
    armT: kind === 'mine' ? 7 : 0,
    elite,
    laserCharge: kind === 'laser_turret' ? 0 : 0,
    laserFireT: kind === 'laser_turret' ? 0 : 0,
    spawnT: kind === 'carrier' ? 2 : 0,
    teleportT: kind === 'phantom' ? 3 : 0,
    phantomAlpha: 1
  });
}

function addFormation(ev, t, kind, n) {
  const sm = 1 + (n - 1) * 0.04;
  let form;
  if (kind === 'tank' || kind === 'dreadnought') form = 'tanks';
  else if (kind === 'mine') form = 'mines';
  else if (kind === 'swarm') form = 'swarm';
  else if (kind === 'carrier' || kind === 'mothership') form = 'big';
  else form = pick(['line', 'v', 'col', 'side']);
  if (form === 'line') {
    const m = 4 + (n > 5 ? 1 : 0) + (n > 50 ? 1 : 0);
    const vy = (kind === 'jelly' ? 55 : kind === 'crab' ? 95 : kind === 'ghost' ? 70 : 80) * sm;
    for (let i = 0; i < m; i++) {
      const x = G.W / (m + 1) * (i + 1);
      ev.push({ t, fn: () => spawnEnemy(kind, x, -40, { vy }) });
    }
  } else if (form === 'v') {
    for (let i = -2; i <= 2; i++) {
      const x = G.W / 2 + i * 55;
      ev.push({ t: t + Math.abs(i) * 0.12, fn: () => spawnEnemy(kind, x, -40 - Math.abs(i) * 26, { vy: 90 * sm, amp: 14 }) });
    }
  } else if (form === 'col') {
    const m = 4 + (n > 30 ? 1 : 0);
    const x = rand(60, G.W - 60);
    for (let i = 0; i < m; i++) {
      const xx = clamp(x + rand(-30, 30), 40, G.W - 40);
      ev.push({ t: t + i * 0.3, fn: () => spawnEnemy(kind, xx, -40, { vy: (kind === 'diver' ? 60 : 110) * sm, amp: kind === 'diver' ? 8 : 40 }) });
    }
  } else if (form === 'side') {
    const dir = pick([-1, 1]);
    for (let i = 0; i < 3; i++) {
      const x = dir < 0 ? -26 : G.W + 26;
      ev.push({ t: t + i * 0.35, fn: () => spawnEnemy(kind, x, 70 + i * 46, { vx: -dir * 105 * sm, vy: 55 * sm, amp: 10 }) });
    }
  } else if (form === 'tanks') {
    ev.push({ t, fn: () => spawnEnemy(kind, G.W * 0.28, -46, { vy: 38 * sm, amp: 6 }) });
    ev.push({ t: t + 0.5, fn: () => spawnEnemy(kind, G.W * 0.72, -46, { vy: 38 * sm, amp: 6 }) });
  } else if (form === 'mines') {
    const m = 2 + (n >= 9 ? 1 : 0) + (n >= 50 ? 1 : 0);
    for (let i = 0; i < m; i++) {
      ev.push({ t: t + i * 0.4, fn: () => spawnEnemy(kind, rand(50, G.W - 50), -40, { vy: 26 }) });
    }
  } else if (form === 'swarm') {
    const m = 8 + Math.floor(n / 10);
    for (let i = 0; i < m; i++) {
      const x = rand(40, G.W - 40);
      ev.push({ t: t + i * 0.1, fn: () => spawnEnemy(kind, x, -40, { vy: 140 * sm, amp: 18 }) });
    }
  } else if (form === 'big') {
    ev.push({ t, fn: () => spawnEnemy(kind, G.W / 2, -60, { vy: 28 * sm, amp: 30 }) });
  }
}

function buildWaves(n) {
  const ev = [];
  let t = 1.5;
  const waves = 10;
  const kinds = levelKinds(n);
  for (let w = 0; w < waves; w++) {
    let kind;
    if (w === 0) kind = 'crab';
    else if (w < 2 && kinds.length > 1) kind = pick(kinds.slice(0, Math.min(2, kinds.length)));
    else if (w === waves - 1 && kinds.length > 2) kind = pick(kinds.slice(-3));
    else kind = pick(kinds);
    addFormation(ev, t, kind, n);
    t += Math.max(1.5, 3.2 - n * 0.015);
  }
  ev.sort((a, b) => a.t - b.t);
  return ev;
}

}

let lastTs = 0, rafId = 0;
function loop(ts) {
  if (!G.running) return;
  rafId = requestAnimationFrame(loop);
  const dt = Math.min(0.033, (ts - lastTs) / 1000 || 0.016);
  lastTs = ts;
  if (!G.paused) update(dt);
  draw();
}
function startLoop() {
  if (G.running) return;
  G.running = true; lastTs = performance.now(); rafId = requestAnimationFrame(loop);
}
function stopLoop() { G.running = false; if (rafId) cancelAnimationFrame(rafId); }

function startLevel(n) {
  G.level = n; G.score = 0; G.time = 0;
  G.enemies = []; G.pbul = []; G.ebul = []; G.pups = []; G.parts = [];
  G.boss = null; G.planets = []; G.shake = 0;
  G.drones = []; G.blackholes = [];
  G.slowT = 0; G.magnetT = 0; G.phoenix = false;
  G.state = 'waves';
  showScreen('game');
  resizeCanvas();
  P.x = G.W / 2; P.y = G.H - 90;
  P.lives = 3; P.gun = 1; P.inv = 1.6; P.shield = 0; P.fireT = 0.3;
  P.laserT = 0; P.rocketT = 0;
  IN.target.x = P.x; IN.target.y = P.y;
  G.events = buildWaves(n);
  renderHearts();
  $('score').textContent = '0';

  $('boss-bar').classList.remove('show');
  $('ov-pause').classList.remove('show');

}

function update(dt) {
  const effectiveDt = G.slowT > 0 ? dt * 0.4 : dt;
  G.time += dt;
  if (G.slowT > 0) G.slowT -= dt;
  if (G.magnetT > 0) G.magnetT -= dt;
  updateBackground(dt);

  if (G.state === 'waves') {
    while (G.events.length && G.events[0].t <= G.time) G.events.shift().fn();
    if (!G.events.length && !G.enemies.length && G.time > 2) {
      G.state = 'bossintro';
      G.bossIntroT = 1.5;
      showBanner('БОСС!', BOSS_NAMES[(G.level - 1) % 10]);
      FX.warn();
    }
  } else if (G.state === 'bossintro') {
    G.bossIntroT -= dt;
    if (G.bossIntroT <= 0) spawnBoss();
  }

  updatePlayer(dt);
  updateEnemies(effectiveDt);
  if (G.boss) updateBoss(effectiveDt);
  updateBullets(effectiveDt);
  updatePups(dt);
  updateParts(dt);
  updateDrones(dt);
  updateBlackholes(dt);
  collide();
  updateHUD();

  if (G.shake > 0) G.shake -= dt;

  if (G.state === 'bossdie') {
    G.bossDieT -= dt;
    if (G.boss && G.boss.expT !== undefined) {
      G.boss.expT -= dt;
      if (G.boss.expT <= 0) {
        G.boss.expT = 0.14;
        explode(G.boss.x + rand(-50, 50), G.boss.y + rand(-30, 30), '#ffd23f', false);
        FX.explode();
      }
    }
    if (G.bossDieT <= 0 && G.boss) {
      explode(G.boss.x, G.boss.y, '#ffd23f', true);
      explode(G.boss.x, G.boss.y, '#ff5a6e', true);
      FX.bigExplode(); haptic('medium');
      G.boss = null; $('boss-bar').classList.remove('show');
      G.state = 'outro'; G.outroT = 1.0;
    }
  } else if (G.state === 'outro') {
    G.outroT -= dt;
    if (G.outroT <= 0) showResult(true);
  } else if (G.state === 'dead') {
    G.deadT -= dt;
    if (G.deadT <= 0) showResult(false);
  }
}

function updateBackground(dt) {
  for (const s of G.stars) {
    s.y += s.v * dt;
    if (s.y > G.H + 4) { s.y = -4; s.x = rand(0, G.W); }
  }
  for (const nb of G.nebulas) {
    nb.y += 4 * dt;
    if (nb.y - nb.r > G.H) { nb.y = -nb.r; nb.x = rand(30, G.W - 30); }
  }
  G.planetT -= dt;
  if (G.planetT <= 0) {
    G.planetT = rand(9, 15);
    const pal = G.palette;
    const r = rand(30, 64);
    G.planets.push({
      x: rand(r + 10, G.W - r - 10), y: -r * 2, r, v: rand(16, 26),
      col: pal.planet[0], col2: pal.planet[1],
      ring: Math.random() < 0.5,
      craters: [[rand(-0.5, 0.5), rand(-0.5, 0.5), rand(0.16, 0.3)], [rand(-0.4, 0.4), rand(-0.3, 0.4), rand(0.1, 0.2)]]
    });
  }
  for (let i = G.planets.length - 1; i >= 0; i--) {
    const p = G.planets[i];
    p.y += p.v * dt;
    if (p.y - p.r > G.H + 20) G.planets.splice(i, 1);
  }
}

function updatePlayer(dt) {
  if (G.state === 'dead') return;
  let kx = 0, ky = 0;
  if (IN.keys.ArrowLeft || IN.keys.KeyA) kx -= 1;
  if (IN.keys.ArrowRight || IN.keys.KeyD) kx += 1;
  if (IN.keys.ArrowUp || IN.keys.KeyW) ky -= 1;
  if (IN.keys.ArrowDown || IN.keys.KeyS) ky += 1;
  if (kx || ky) {
    IN.target.x = clamp(IN.target.x + kx * 320 * dt, 26, G.W - 26);
    IN.target.y = clamp(IN.target.y + ky * 320 * dt, 90, G.H - 50);
  }
  const tx = clamp(IN.target.x, 26, G.W - 26);
  const ty = clamp(IN.target.y, 90, G.H - 50);
  const k = Math.min(1, dt * 14);
  P.x += (tx - P.x) * k;
  P.y += (ty - P.y) * k;
  P.tilt = clamp((tx - P.x) * 0.02, -0.3, 0.3);

  if (P.inv > 0) P.inv -= dt;
  if (P.shield > 0) P.shield -= dt;
  if (P.laserT > 0) P.laserT -= dt;
  if (P.rocketT > 0) P.rocketT -= dt;

  P.fireT -= dt;
  if (P.fireT <= 0 && (G.state === 'waves' || G.state === 'boss' || G.state === 'bossintro')) {
    const y = P.y - 28;
    if (P.laserT > 0) {
      G.pbul.push({ x: P.x, y: y, vx: 0, vy: -1200, r: 12, kind: 'laser', life: 0.3 });
      P.fireT = 0.05;
      FX.laser();
    } else if (P.rocketT > 0) {
      addRocket(P.x, y, 0, -500);
      P.fireT = 0.4;
      FX.explode();
    } else {
      let gun = P.gun;
      if (P.multishotT > 0) gun = Math.min(3, gun + 1);
      if (gun === 1) {
        addPB(P.x, y, 0, -640);
        P.fireT = 0.17;
      } else if (gun === 2) {
        addPB(P.x - 12, y + 6, 0, -640);
        addPB(P.x + 12, y + 6, 0, -640);
        P.fireT = 0.155;
      } else {
        addPB(P.x, y - 2, 0, -650);
        addPB(P.x - 14, y + 8, -95, -620);
        addPB(P.x + 14, y + 8, 95, -620);
        P.fireT = 0.14;
      }
      FX.shoot();
    }
  }
}

function updateEnemies(dt) {
  const n = G.level;
  for (let i = G.enemies.length - 1; i >= 0; i--) {
    const e = G.enemies[i];
    e.t += dt;
    if (e.hitT > 0) e.hitT -= dt;

    if (e.kind === 'crab' || e.kind === 'jelly' || e.kind === 'ghost' || e.kind === 'elite') {
      e.y += e.vy * dt;
      e.x = e.baseX + Math.sin(e.t * (e.kind === 'crab' || e.kind === 'elite' ? 2.3 : 1.6)) * e.amp + e.vx * dt;
      if (e.vx) e.baseX += e.vx * dt;
      if ((e.kind === 'jelly' || e.kind === 'ghost') && e.y > 30 && e.y < G.H - 200) {
        e.fireT -= dt;
        if (e.fireT <= 0) {
          aimEB(e.x, e.y + 14, 150 + n * 3, 0.08);
          e.fireT = rand(2.6, 3.6) * Math.max(0.7, 1 - n * 0.02);
        }
      }
    } else if (e.kind === 'tank' || e.kind === 'dreadnought') {
      e.y += e.vy * dt;
      e.x = e.baseX + Math.sin(e.t * 0.8) * e.amp;
      if (e.y > 40) {
        e.fireT -= dt;
        if (e.fireT <= 0) {
          const count = e.kind === 'dreadnought' ? 7 : (n >= 7 ? 5 : 3);
          fanEB(e.x, e.y + 20, count, 145 + n * 3);
          e.fireT = rand(2.1, 2.9) * Math.max(0.72, 1 - n * 0.02);
        }
      }
    } else if (e.kind === 'diver') {
      if (e.state === 0) {
        e.y += 24 * dt;
        e.x = e.baseX + Math.sin(e.t * 3) * 12;
        e.aimT -= dt;
        if (e.aimT <= 0) {
          e.state = 1;
          const ang = Math.atan2(P.y - e.y, P.x - e.x);
          const sp = 330 + n * 6;
          e.vx = Math.cos(ang) * sp; e.vy = Math.sin(ang) * sp;
        }
      } else {
        e.x += e.vx * dt; e.y += e.vy * dt;
      }
    } else if (e.kind === 'mine') {
      e.y += e.vy * dt;
      e.x += (G.W / 2 - e.baseX) * 0.06 * dt;
      e.armT -= dt;
      if (e.armT <= 0 || (G.state !== 'dead' && dist2(e.x, e.y, P.x, P.y) < sq(62))) {
        for (let k = 0; k < 6; k++) {
          const a = k / 6 * TAU + 0.4;
          G.ebul.push({ x: e.x, y: e.y, vx: Math.cos(a) * 135, vy: Math.sin(a) * 135, r: 7 });
        }
        explode(e.x, e.y, '#9aa3b5', false); FX.explode();
        G.enemies.splice(i, 1); continue;
      }
    } else if (e.kind === 'asteroid') {
      e.y += e.vy * dt;
      e.x += e.vx * dt + Math.sin(e.t * 1.5) * 20 * dt;
    } else if (e.kind === 'sniper') {
      e.y += e.vy * dt;
      e.x = e.baseX + Math.sin(e.t * 1.2) * 30;
      if (e.y > 60 && e.y < G.H - 200) {
        e.fireT -= dt;
        if (e.fireT <= 0) {
          G.ebul.push({
            x: e.x, y: e.y + 20,
            vx: (P.x - e.x) * 1.8, vy: (P.y - e.y) * 1.8,
            r: 8, sniper: true
          });
          FX.eshoot();
          e.fireT = rand(1.8, 2.8) * Math.max(0.7, 1 - n * 0.02);
        }
      }
    } else if (e.kind === 'carrier') {
      e.y += e.vy * dt;
      e.x = e.baseX + Math.sin(e.t * 0.5) * e.amp;
      e.spawnT -= dt;
      if (e.spawnT <= 0 && e.y > 40 && e.y < G.H - 200) {
        for (let k = 0; k < 3; k++) {
          spawnEnemy('swarm', e.x + rand(-10, 10), e.y + 20, { vy: 140, amp: 20 });
        }
        e.spawnT = 4;
      }
    } else if (e.kind === 'laser_turret') {
      e.y += e.vy * dt;
      e.x = e.baseX + Math.sin(e.t * 0.7) * 20;
      if (e.y > 50 && e.y < G.H - 180) {
        e.laserCharge += dt;
        e.laserFireT -= dt;
        if (e.laserCharge > 1.5) {
          for (let k = 0; k < 4; k++) {
            G.ebul.push({
              x: e.x, y: e.y - 20,
              vx: 0, vy: 380 + k * 40,
              r: 6, laser: true
            });
          }
          FX.eshoot();
          e.laserCharge = 0;
        }
      }
    } else if (e.kind === 'swarm') {
      e.y += e.vy * dt;
      e.x = e.baseX + Math.sin(e.t * 5 + i) * e.amp + Math.cos(e.t * 3) * 12 * dt;
    } else if (e.kind === 'mothership') {
      e.y += e.vy * dt;
      e.x = e.baseX + Math.sin(e.t * 0.3) * e.amp;
      e.fireT -= dt;
      if (e.fireT <= 0 && e.y > 60) {
        const cnt = 12;
        for (let k = 0; k < cnt; k++) {
          const a = k / cnt * TAU + e.t;
          G.ebul.push({ x: e.x, y: e.y + 16, vx: Math.cos(a) * 140, vy: Math.sin(a) * 140, r: 7 });
        }
        FX.eshoot();
        e.fireT = 3.5;
      }
    } else if (e.kind === 'phantom') {
      e.y += e.vy * dt;
      e.x = e.baseX + Math.sin(e.t * 2) * e.amp;
      e.teleportT -= dt;
      e.phantomAlpha = 0.5 + 0.5 * Math.sin(e.t * 3);
      if (e.teleportT <= 0 && e.y > 40 && e.y < G.H - 200) {
        e.baseX = rand(60, G.W - 60);
        e.x = e.baseX;
        e.teleportT = rand(2, 4);
        for (let k = 0; k < 5; k++) {
          const a = k / 5 * TAU;
          G.ebul.push({ x: e.x, y: e.y, vx: Math.cos(a) * 130, vy: Math.sin(a) * 130, r: 7 });
        }
        FX.eshoot();
      }
    } else if (e.kind === 'overlord') {
      e.y += e.vy * dt;
      e.x = e.baseX + Math.sin(e.t * 0.8) * e.amp;
      e.fireT -= dt;
      if (e.fireT <= 0 && e.y > 50 && e.y < G.H - 180) {
        fanEB(e.x, e.y + 20, 5, 170);
        e.fireT = rand(1.8, 2.5);
      }
    }
    
    if (e.y > G.H + 60 || e.x < -80 || e.x > G.W + 80) G.enemies.splice(i, 1);
  }
}

function spawnBoss() {
  const n = G.level;
  const kind = (n - 1) % 10;
  const hp = 40 + n * 10 + Math.floor(n / 10) * 20;
  G.boss = {
    kind, name: BOSS_NAMES[kind], hp, maxHp: hp,
    x: G.W / 2, y: -70, t: 0, atkT: 1.4, atkIdx: 0,
    dying: 0, hitT: 0, r: 50, expT: 0
  };
  G.state = 'boss';
  $('boss-name').textContent = G.boss.name;
  $('boss-hp').firstElementChild.style.width = '100%';
  $('boss-bar').classList.add('show');
}
function updateBoss(dt) {
  const b = G.boss;
  const n = G.level;
  b.t += dt;
  if (b.hitT > 0) b.hitT -= dt;
  if (b.dying) return;

  if (b.y < 118) b.y += (118 - b.y) * Math.min(1, dt * 1.8);
  else b.x = G.W / 2 + Math.sin(b.t * 0.55) * (G.W / 2 - 78);

  b.atkT -= dt;
  if (b.atkT <= 0 && G.state === 'boss') {
    const seq = ['aim', 'fan'];
    if (n >= 3) seq.push('ring');
    if (n >= 5) seq.push('summon');
    if (n >= 20) seq.push('cross');
    if (n >= 50) seq.push('spiral');
    const atk = seq[b.atkIdx++ % seq.length];
    if (atk === 'aim') {
      const c = n >= 8 ? 5 : 3;
      for (let i = 0; i < c; i++) aimEB(b.x, b.y + 34, 215 + n * 4, 0.12);
    } else if (atk === 'fan') {
      fanEB(b.x, b.y + 34, n >= 9 ? 7 : 5, 175 + n * 3);
    } else if (atk === 'ring') {
      const cnt = 10;
      for (let i = 0; i < cnt; i++) {
        const a = i / cnt * TAU + b.t;
        G.ebul.push({ x: b.x, y: b.y, vx: Math.cos(a) * (132 + n * 2), vy: Math.sin(a) * (132 + n * 2), r: 7 });
      }
      FX.eshoot();
    } else if (atk === 'summon') {
      for (let i = 0; i < 2; i++) spawnEnemy('crab', b.x + rand(-70, 70), b.y + 40, { vy: 110 });
    } else if (atk === 'cross') {
      for (let i = 0; i < 8; i++) {
        const a = i / 8 * TAU + b.t;
        G.ebul.push({ x: b.x, y: b.y, vx: Math.cos(a) * 180, vy: Math.sin(a) * 180, r: 8 });
      }
      FX.eshoot();
    } else if (atk === 'spiral') {
      for (let i = 0; i < 16; i++) {
        const a = i / 16 * TAU + b.t * 2;
        G.ebul.push({ x: b.x, y: b.y, vx: Math.cos(a) * 150, vy: Math.sin(a) * 150, r: 6 });
      }
      FX.eshoot();
    }
    const low = b.hp / b.maxHp < 0.35;
    b.atkT = (low ? 0.85 : 1.45) * Math.max(0.72, 1 - n * 0.015);
  }
}
function hitBoss() {
  const b = G.boss;
  b.hp--; b.hitT = 0.06;
  $('boss-hp').firstElementChild.style.width = Math.max(0, b.hp / b.maxHp * 100) + '%';
  if (b.hp <= 0 && !b.dying) {
    b.dying = 1; G.state = 'bossdie'; G.bossDieT = 1.4; b.expT = 0;
    const pts = 1500 + G.level * 250;
    addScore(pts);
    addTextPart(b.x, b.y - 20, '+' + pts, '#ffd23f');
    haptic('success');
  }
}

function updateBullets(dt) {
  for (let i = G.pbul.length - 1; i >= 0; i--) {
    const b = G.pbul[i];
    b.x += b.vx * dt; b.y += b.vy * dt;
    if (b.life !== undefined) { b.life -= dt; if (b.life <= 0) { G.pbul.splice(i, 1); continue; } }
    if (b.y < -30 || b.x < -20 || b.x > G.W + 20) G.pbul.splice(i, 1);
  }
  for (let i = G.ebul.length - 1; i >= 0; i--) {
    const b = G.ebul[i];
    b.x += b.vx * dt; b.y += b.vy * dt;
    if (b.y > G.H + 30 || b.y < -60 || b.x < -30 || b.x > G.W + 30) G.ebul.splice(i, 1);
  }
}

function updatePups(dt) {
  for (let i = G.pups.length - 1; i >= 0; i--) {
    const u = G.pups[i];
    u.t += dt;
    if (G.magnetT > 0) {
      const dx = P.x - u.x, dy = P.y - u.y;
      const d = Math.hypot(dx, dy) || 1;
      const pull = 400;
      u.x += (dx / d) * pull * dt;
      u.y += (dy / d) * pull * dt;
    } else {
      u.y += 78 * dt;
      u.x += Math.sin(u.t * 3) * 18 * dt;
    }
    if (u.y > G.H + 30) { G.pups.splice(i, 1); continue; }
    if (G.state !== 'dead' && dist2(u.x, u.y, P.x, P.y) < sq(34)) {
      collectPup(u.kind);
      G.pups.splice(i, 1);
    }
  }
}

    else { addScore(150); addTextPart(P.x, P.y - 40, '+150', '#ffd23f'); }

    P.shield = 6;
    addTextPart(P.x, P.y - 40, 'ЩИТ', '#4cc9f0');
  } else if (kind === 'star') {
    addScore(250);
    addTextPart(P.x, P.y - 40, '+250', '#ffd23f');
  } else if (kind === 'bomb') {
    bombEffect(false);
    addTextPart(P.x, P.y - 40, 'БОМБА!', '#ff3838');
    FX.bomb(); haptic('medium');
  } else if (kind === 'magnet') {
    G.magnetT = 10;
    addTextPart(P.x, P.y - 40, 'МАГНИТ', '#b388ff');
  } else if (kind === 'slow') {
    G.slowT = 6;
    addTextPart(P.x, P.y - 40, 'ЗАМЕДЛЕНИЕ', '#80deea');
  } else if (kind === 'drone') {
    spawnDrone();
    addTextPart(P.x, P.y - 40, 'ДРОН', '#69f0ae');
  } else if (kind === 'rocket') {
    P.rocketT = 8;
    addTextPart(P.x, P.y - 40, 'РАКЕТЫ', '#ffab40');
  } else if (kind === 'laser') {
    P.laserT = 6;

  } else if (kind === 'blackhole') {
    spawnBlackhole();

    P.multishotT = 10;
    addTextPart(P.x, P.y - 40, 'МУЛЬТИ-ВЫСТРЕЛ', '#e1bee7');
  } else if (kind === 'phoenix') {
    G.phoenix = true;
    addTextPart(P.x, P.y - 40, 'ФЕНИКС', '#ffb300');
    FX.phoenix();
  } else if (kind === 'nuke') {
    bombEffect(true);
    addTextPart(P.x, P.y - 40, 'ЯДЕРНЫЙ УДАР!', '#d50000');
    FX.bomb(); haptic('heavy');
  }
}

function bombEffect(nuke) {
  const count = G.enemies.length;
  for (let i = count - 1; i >= 0; i--) {
    const e = G.enemies[i];
    explode(e.x, e.y, E_COL[e.kind] || '#ff3838', true);
    addScore(e.pts);
    if (Math.random() < e.drop * 0.5) spawnPup(e.x, e.y);
  }
  G.enemies = [];
  G.ebul = [];
  explode(G.W / 2, G.H / 2, nuke ? '#ffd23f' : '#ff3838', true);
  explode(P.x, P.y, '#fff', true);
  G.shake = 0.8;
  if (nuke && G.boss && !G.boss.dying) {
    G.boss.hp = Math.max(0, G.boss.hp - 20);
    $('boss-hp').firstElementChild.style.width = Math.max(0, G.boss.hp / G.boss.maxHp * 100) + '%';
    if (G.boss.hp <= 0) hitBoss();
  }
}

  });

function spawnBlackhole() {
  G.blackholes.push({
    x: G.W / 2, y: G.H / 2,
    t: 0, life: 5, r: 0, maxR: 120
  });
}

function updateBlackholes(dt) {
  for (let i = G.blackholes.length - 1; i >= 0; i--) {
    const bh = G.blackholes[i];
    bh.t += dt;
    bh.life -= dt;
    bh.r = Math.min(bh.maxR, bh.r + 200 * dt);
    if (bh.life <= 0) { G.blackholes.splice(i, 1); continue; }
    for (let j = G.enemies.length - 1; j >= 0; j--) {
      const e = G.enemies[j];
      const dx = bh.x - e.x, dy = bh.y - e.y;
      const d = Math.hypot(dx, dy) || 1;
      const force = 300;
      e.x += (dx / d) * force * dt;
      e.y += (dy / d) * force * dt;
      if (d < bh.r * 0.5) {
        explode(e.x, e.y, E_COL[e.kind] || '#5e35b1', false);
        addScore(e.pts);
        G.enemies.splice(j, 1);
      }
    }
    for (let j = G.ebul.length - 1; j >= 0; j--) {
      const b = G.ebul[j];
      const dx = bh.x - b.x, dy = bh.y - b.y;
      const d = Math.hypot(dx, dy) || 1;
      const force = 400;
      b.x += (dx / d) * force * dt;
      b.y += (dy / d) * force * dt;
      if (d < bh.r * 0.5) G.ebul.splice(j, 1);
    }
  }
}

function drawBlackhole(c, bh) {
  c.save(); c.translate(bh.x, bh.y);
  for (let i = 0; i < 3; i++) {
    const r = bh.r - i * 15;
    const a = 0.2 + i * 0.15;
    c.strokeStyle = 'rgba(186,104,200,' + a + ')';
    c.lineWidth = 4;
    c.beginPath(); c.arc(0, 0, r, 0, TAU); c.stroke();
  }
  c.fillStyle = '#000';
  c.beginPath(); c.arc(0, 0, bh.r * 0.4, 0, TAU); c.fill();
  c.strokeStyle = 'rgba(255,210,63,.6)'; c.lineWidth = 2;
  c.beginPath();
  for (let a = 0; a < TAU * 3; a += 0.2) {
    const r = bh.r * 0.4 + a * 3;
    const x = Math.cos(a + bh.t * 3) * r;
    const y = Math.sin(a + bh.t * 3) * r;
    if (a === 0) c.moveTo(x, y); else c.lineTo(x, y);
    if (r > bh.r) break;
  }
  c.stroke();
  c.restore();
}

function meteorShower() {
  const count = 6 + Math.floor(G.level / 10);
  for (let i = 0; i < count; i++) {
    setTimeout(() => {
      const x = rand(50, G.W - 50);
      const y = -50;
      G.pbul.push({
        x, y, vx: rand(-30, 30), vy: 450,
        r: 14, kind: 'meteor', life: 2
      });
      FX.meteor();
    }, i * 200);
  }
}

function updateParts(dt) {
  for (let i = G.parts.length - 1; i >= 0; i--) {
    const p = G.parts[i];
    p.life -= dt;
    if (p.life <= 0) { G.parts.splice(i, 1); continue; }
    if (p.vx !== undefined) { p.x += p.vx * dt; p.y += p.vy * dt; p.vx *= (1 - 1.6 * dt); }
    if (p.vy !== undefined && p.type === 'text') p.y += p.vy * dt;
  }
}

function addScore(v) { G.score += v; }

function killEnemy(idx, givePts) {
  const e = G.enemies[idx];
  explode(e.x, e.y, E_COL[e.kind] || '#fff', e.kind === 'tank' || e.kind === 'dreadnought' || e.kind === 'mothership');
  FX.explode();
  if (givePts) {
    addScore(e.pts);
    if (Math.random() < e.drop) spawnPup(e.x, e.y);
  }
  G.enemies.splice(idx, 1);
}

function damagePlayer() {
  if (P.inv > 0 || P.shield > 0 || G.state === 'dead') return;
  P.lives--;
  renderHearts();
  P.inv = 2.2;
  P.gun = Math.max(1, P.gun - 1);
  G.shake = 0.4;
  FX.hurt(); haptic('error');
  explode(P.x, P.y, '#f8f9ff', false);
  if (P.lives <= 0) {
    if (G.phoenix) {
      G.phoenix = false;
      P.lives = 1;
      renderHearts();
      P.inv = 3;
      addTextPart(P.x, P.y - 50, 'ФЕНИКС!', '#ffb300');
      FX.phoenix();
      explode(P.x, P.y, '#ffb300', true);
      return;
    }
    G.state = 'dead';
    G.deadT = 1.25;
    explode(P.x, P.y, '#ffd23f', true);
    explode(P.x, P.y, '#ff5a6e', true);
    FX.bigExplode();
  }
}

function collide() {
  for (let i = G.pbul.length - 1; i >= 0; i--) {
    const b = G.pbul[i];
    let dead = false;
    if (G.boss && !G.boss.dying && dist2(b.x, b.y, G.boss.x, G.boss.y) < sq(G.boss.r + 6)) {
      const dmg = b.kind === 'rocket' ? 5 : b.kind === 'laser' ? 2 : b.kind === 'meteor' ? 3 : 1;
      for (let k = 0; k < dmg; k++) hitBoss();
      FX.hit();
      dead = true;
    } else {
      for (let j = G.enemies.length - 1; j >= 0; j--) {
        const e = G.enemies[j];
        if (dist2(b.x, b.y, e.x, e.y) < sq(e.r + 7)) {
          const dmg = b.kind === 'rocket' ? 4 : b.kind === 'laser' ? 2 : b.kind === 'meteor' ? 2 : 1;
          e.hp -= dmg;
          e.hitT = 0.07;
          FX.hit();
          if (e.hp <= 0) killEnemy(j, true);
          dead = true;
          break;
        }
      }
    }
    if (dead && b.kind !== 'laser') G.pbul.splice(i, 1);
  }
  if (G.state === 'dead') return;
  for (let i = G.ebul.length - 1; i >= 0; i--) {
    const b = G.ebul[i];
    if (dist2(b.x, b.y, P.x, P.y) < sq(b.r + 14)) {
      if (P.shield > 0) { G.ebul.splice(i, 1); explode(b.x, b.y, '#4cc9f0', false); continue; }
      if (P.inv > 0) continue;
      G.ebul.splice(i, 1);
      damagePlayer();
    }
  }
  for (let j = G.enemies.length - 1; j >= 0; j--) {
    const e = G.enemies[j];
    if (dist2(e.x, e.y, P.x, P.y) < sq(e.r + 15)) {
      if (P.inv > 0) continue;
      if (e.kind === 'mine') { killEnemy(j, false); damagePlayer(); }
      else if (P.shield > 0) { killEnemy(j, true); }
      else { killEnemy(j, false); damagePlayer(); }
    }
  }
  if (G.boss && !G.boss.dying && P.inv <= 0 && P.shield <= 0 &&
      dist2(G.boss.x, G.boss.y, P.x, P.y) < sq(G.boss.r + 16)) {
    damagePlayer();
  }
}

function updateHUD() {
  $('score').textContent = fmt(G.score);
}

function draw() {
  const k = DPR * SCALE;
  ctx.setTransform(k, 0, 0, k, 0, 0);
  if (G.shake > 0) ctx.translate(rand(-5, 5) * G.shake * 2, rand(-5, 5) * G.shake * 2);
  
  const pal = G.palette;
  ctx.fillStyle = pal.bg;
  ctx.fillRect(-10, -10, G.W + 20, G.H + 20);
  
  const grad = ctx.createLinearGradient(0, 0, 0, G.H);
  grad.addColorStop(0, pal.bgGrad);
  grad.addColorStop(1, pal.bg);
  ctx.fillStyle = grad;
  ctx.fillRect(-10, -10, G.W + 20, G.H + 20);

function drawPup(u) {
  const col = BONUS_TYPES[u.kind]?.color || '#f8f9ff';
  const pul = 0.5 + 0.5 * Math.sin(u.t * 5);
  ctx.strokeStyle = 'rgba(244,246,255,' + (0.25 + 0.3 * pul) + ')';
  ctx.lineWidth = 2;
  ctx.beginPath(); ctx.arc(u.x, u.y, 17 + pul * 3, 0, TAU); ctx.stroke();
  ctx.fillStyle = col; ctx.strokeStyle = INK; ctx.lineWidth = 3;
  ctx.beginPath(); ctx.arc(u.x, u.y, 15, 0, TAU); ctx.fill(); ctx.stroke();
  ctx.strokeStyle = '#fff'; ctx.fillStyle = '#fff'; ctx.lineWidth = 2.6; ctx.lineCap = 'round'; ctx.lineJoin = 'round';
  ctx.save(); ctx.translate(u.x, u.y);
  if (u.kind === 'heart') {
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.moveTo(0, 5.5);
    ctx.bezierCurveTo(-8, -1, -5.5, -7.5, 0, -3.5);
    ctx.bezierCurveTo(5.5, -7.5, 8, -1, 0, 5.5);
    ctx.fill();
  } else if (u.kind === 'gun') {
    ctx.beginPath();
    ctx.moveTo(1.5, -7); ctx.lineTo(-4.5, 1.5); ctx.lineTo(-0.5, 1.5); ctx.lineTo(-1.5, 7); ctx.lineTo(4.5, -1.5); ctx.lineTo(0.5, -1.5);
    ctx.closePath(); ctx.fill();
  } else if (u.kind === 'shield') {
    ctx.beginPath();
    ctx.moveTo(0, -7); ctx.lineTo(5.5, -4.5); ctx.lineTo(5.5, 1.5); ctx.quadraticCurveTo(5.5, 6, 0, 7.5); ctx.quadraticCurveTo(-5.5, 6, -5.5, 1.5); ctx.lineTo(-5.5, -4.5);
    ctx.closePath(); ctx.fill();
  } else if (u.kind === 'star') {
    ctx.beginPath();
    for (let i = 0; i < 5; i++) {
      const a = -Math.PI / 2 + i * 1.2566;
      ctx.lineTo(Math.cos(a) * 7.5, Math.sin(a) * 7.5);
      const a2 = a + 0.6283;
      ctx.lineTo(Math.cos(a2) * 3.2, Math.sin(a2) * 3.2);
    }
    ctx.closePath(); ctx.fill();
  } else if (u.kind === 'bomb') {
    ctx.fillStyle = '#263238';
    ctx.beginPath(); ctx.arc(0, 2, 8, 0, TAU); ctx.fill();
    ctx.strokeStyle = '#fff'; ctx.lineWidth = 2;
    ctx.beginPath(); ctx.moveTo(0, -6); ctx.lineTo(3, -10); ctx.stroke();
    ctx.fillStyle = '#ffd23f';
    ctx.beginPath(); ctx.arc(3, -10, 2, 0, TAU); ctx.fill();
  } else if (u.kind === 'magnet') {
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(0, 2, 7, 0, Math.PI, false);
    ctx.lineTo(-7, -4); ctx.lineTo(-3, -4); ctx.lineTo(-3, 2);
    ctx.arc(0, 2, 3, Math.PI, 0, true);
    ctx.lineTo(3, -4); ctx.lineTo(7, -4);
    ctx.closePath(); ctx.fill();
    ctx.fillStyle = '#d32f2f';
    ctx.fillRect(-7, -6, 4, 2); ctx.fillRect(3, -6, 4, 2);
  } else if (u.kind === 'slow') {
    ctx.strokeStyle = '#fff'; ctx.lineWidth = 2;
    ctx.beginPath(); ctx.arc(0, 0, 8, 0, TAU); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(0, 0); ctx.lineTo(0, -5); ctx.moveTo(0, 0); ctx.lineTo(4, 2); ctx.stroke();
  } else if (u.kind === 'drone') {
    ctx.fillStyle = '#fff';
    ctx.beginPath(); ctx.arc(0, 0, 6, 0, TAU); ctx.fill();
    ctx.fillStyle = '#263238';
    ctx.beginPath(); ctx.arc(-2, -2, 1.5, 0, TAU); ctx.fill();
    ctx.strokeStyle = '#fff'; ctx.lineWidth = 1.5;
    ctx.beginPath(); ctx.moveTo(-6, -6); ctx.lineTo(6, -6); ctx.stroke();
  } else if (u.kind === 'rocket') {
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.moveTo(0, -8); ctx.lineTo(4, 0); ctx.lineTo(3, 7); ctx.lineTo(-3, 7); ctx.lineTo(-4, 0);
    ctx.closePath(); ctx.fill();
    ctx.fillStyle = '#ff6f00';
    ctx.beginPath(); ctx.moveTo(-2, 7); ctx.lineTo(2, 7); ctx.lineTo(0, 10); ctx.closePath(); ctx.fill();
  } else if (u.kind === 'laser') {
    ctx.fillStyle = '#fff';
    rr(ctx, -7, -3, 14, 6, 2); ctx.fill();
    ctx.fillStyle = '#f06292';
    ctx.beginPath(); ctx.arc(7, 0, 3, 0, TAU); ctx.fill();
  } else if (u.kind === 'blackhole') {
    ctx.fillStyle = '#1a1a2e';
    ctx.beginPath(); ctx.arc(0, 0, 7, 0, TAU); ctx.fill();
    ctx.strokeStyle = '#ba68c8'; ctx.lineWidth = 1.5;
    ctx.beginPath();
    for (let a = 0; a < TAU * 2; a += 0.3) {
      const r = 2 + a * 0.8;
      const x = Math.cos(a + u.t * 3) * r, y = Math.sin(a + u.t * 3) * r;
      if (a === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
    }
    ctx.stroke();
  } else if (u.kind === 'meteor') {
    ctx.fillStyle = '#fff';
    ctx.beginPath(); ctx.arc(0, 0, 6, 0, TAU); ctx.fill();
    ctx.fillStyle = '#ff6f00';
    ctx.beginPath(); ctx.moveTo(0, -7); ctx.lineTo(2, -10); ctx.lineTo(-1, -8); ctx.closePath(); ctx.fill();
  } else if (u.kind === 'multishot') {
    for (const s of [-1, 0, 1]) {
      ctx.save(); ctx.rotate(s * 0.3);
      ctx.beginPath(); ctx.moveTo(0, -8); ctx.lineTo(-3, -2); ctx.lineTo(3, -2); ctx.closePath(); ctx.fill();
      ctx.fillRect(-1, -2, 2, 6);
      ctx.restore();
    }
  } else if (u.kind === 'phoenix') {
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.moveTo(0, -8);
    ctx.quadraticCurveTo(-8, -2, -5, 4);
    ctx.quadraticCurveTo(0, 0, 5, 4);
    ctx.quadraticCurveTo(8, -2, 0, -8);
    ctx.fill();
  } else if (u.kind === 'nuke') {
    ctx.fillStyle = '#fff';
    ctx.beginPath(); ctx.arc(0, 0, 8, 0, TAU); ctx.fill();
    ctx.fillStyle = '#263238';
    ctx.beginPath(); ctx.arc(0, 0, 3, 0, TAU); ctx.fill();
    for (let i = 0; i < 3; i++) {
      const a = i / 3 * TAU - Math.PI / 2;
      ctx.beginPath();
      ctx.arc(0, 0, 8, a - 0.4, a + 0.4);
      ctx.lineTo(0, 0);
      ctx.closePath(); ctx.fill();
    }
  }
  ctx.restore();
}
function drawPart(p) {
  const a = clamp(p.life / p.maxLife, 0, 1);
  if (p.type === 'flash') {
    const r = p.r + (p.r1 - p.r) * (1 - a);
    ctx.fillStyle = p.col;
    ctx.globalAlpha = a * 0.5;
    ctx.beginPath(); ctx.arc(p.x, p.y, r, 0, TAU); ctx.fill();
    ctx.globalAlpha = a * 0.9;
    ctx.strokeStyle = '#fff'; ctx.lineWidth = 3;
    ctx.beginPath(); ctx.arc(p.x, p.y, r, 0, TAU); ctx.stroke();
    ctx.globalAlpha = 1;
  } else if (p.type === 'text') {
    ctx.globalAlpha = a;
    ctx.font = '900 17px ' + FONT;
    ctx.textAlign = 'center';
    ctx.lineWidth = 4; ctx.strokeStyle = INK;
    ctx.strokeText(p.str, p.x, p.y);
    ctx.fillStyle = p.col;
    ctx.fillText(p.str, p.x, p.y);
    ctx.globalAlpha = 1;
  } else {
    ctx.globalAlpha = a;
    ctx.fillStyle = p.col;
    ctx.beginPath(); ctx.arc(p.x, p.y, p.r * a + 1, 0, TAU); ctx.fill();
    ctx.globalAlpha = 1;
  }
}

function starsFor(livesLeft) {
  if (livesLeft >= 3) return 3;
  if (livesLeft === 2) return 2;
  return 1;
}
function showResult(win) {
  stopLoop(); stopMusic();
  $('ov-pause').classList.remove('show');
  const n = G.level;
  const final = win && n === TOTAL_LEVELS;
  if (win) {
    const bonus = P.lives * 500;
    if (bonus > 0) addScore(bonus);
  }
  $('res-title').textContent = win ? (final ? 'Галактика спасена!' : 'Уровень пройден!') : 'Корабль подбит!';
  const starsWrap = $('res-stars');
  starsWrap.innerHTML = '';
  if (win && !final) {
    const st = starsFor(P.lives);
    for (let i = 0; i < 3; i++) {
      const s = document.createElement('span');
      s.innerHTML = i < st ? starSvgFull : starSvgEmpty;
      const svg = s.firstElementChild;
      starsWrap.appendChild(svg);
      if (i < st) setTimeout(() => { svg.classList.add('pop'); FX.star(); }, 350 + i * 280);
      else setTimeout(() => svg.classList.add('pop'), 350 + st * 280);
    }
  }
  $('res-score-v').textContent = fmt(G.score);
  const prevBest = SAVE.best[n] || 0;
  const isRec = win && G.score > prevBest;
  $('res-record').classList.toggle('show', isRec);
  if (isRec) metrika('newRecord', { level: n });
  if (win) {
    SAVE.unlocked = Math.max(SAVE.unlocked, Math.min(TOTAL_LEVELS, n + 1));
    SAVE.stars[n] = Math.max(SAVE.stars[n] || 0, starsFor(P.lives));
    SAVE.best[n] = Math.max(prevBest, G.score);
    persist();
    metrika('gameFinish', { level: n, final });
  }
  
  /* ============ СОХРАНЕНИЕ СЧЁТА В ЛИДЕРБОРД (И ПРИ ПОБЕДЕ, И ПРИ ПОРАЖЕНИИ) ============ */
  if (window.GameSDK && typeof GameSDK.saveScore === 'function') {
    const scoreToSend = totalBest();
    console.log('[Cosmo-Boom] Отправка счёта в лидерборд:', scoreToSend);
    GameSDK.saveScore(scoreToSend).then(res => {
      console.log('[Cosmo-Boom] saveScore ответ:', res);
      if (res && res.ok && res.is_new_record) {
        toast('🏆 Новый рекорд в рейтинге!');
      }
    }).catch(e => {
      console.warn('[Cosmo-Boom] saveScore ошибка:', e);
    });
  }
  
  $('res-next').style.display = (win && !final) ? '' : 'none';
  $('res-retry').style.display = final ? 'none' : '';
  $('res-menu').classList.toggle('btn-yellow', final);
  if (final) { FX.finalWin(); SAVE.finalSeen = true; persist(); }
  else if (win) FX.win();
  else FX.lose();
  $('ov-result').classList.add('show');
}

function renderLeaderboard(players) {
  const list = $('lb-list');
  list.innerHTML = '';
  if (!players || players.length === 0) {
    const empty = document.createElement('div');
    empty.className = 'lb-empty';
    empty.textContent = 'Пока нет рекордов. Будьте первым, кто покорит галактику!';
    list.appendChild(empty);
    return;
  }
  players.forEach((p, i) => {
    const row = document.createElement('div');
    row.className = 'lb-row';
    if (i === 0) row.classList.add('top-1');
    else if (i === 1) row.classList.add('top-2');
    else if (i === 2) row.classList.add('top-3');
    // Подсветка "меня" если совпадает ID
    try {
      const u = window.WebApp && window.WebApp.initDataUnsafe && window.WebApp.initDataUnsafe.user;
      if (u && p.id && String(p.id) === String(u.id)) row.classList.add('me');
    } catch (e) {}

    // Медаль / место
    const medal = document.createElement('div');
    medal.className = 'lb-medal';
    if (i === 0) { medal.classList.add('gold'); medal.textContent = '🥇'; }
    else if (i === 1) { medal.classList.add('silver'); medal.textContent = '🥈'; }
    else if (i === 2) { medal.classList.add('bronze'); medal.textContent = '🥉'; }
    else { medal.textContent = '#' + (i + 1); }
    row.appendChild(medal);

    // Аватарка
    const avatar = document.createElement('div');
    avatar.className = 'lb-avatar';
    const displayName = p.name || p.nickname || 'Космонавт';
    const initial = displayName.charAt(0).toUpperCase();
    // Цвет аватара по хешу имени
    const colors = ['#4cc9f0', '#ff5a6e', '#ffd23f', '#7be0d6', '#a97bff', '#ffa53f', '#58c96b', '#ffab40'];
    const colorIdx = displayName.split('').reduce((a, c) => a + c.charCodeAt(0), 0) % colors.length;
    avatar.style.background = colors[colorIdx];
    
    if (p.photo_url) {
      const img = document.createElement('img');
      img.src = p.photo_url;
      img.alt = '';
      img.onerror = function() {
        this.remove();
        avatar.textContent = initial;
        avatar.style.color = '#fff';
      };
      avatar.appendChild(img);
      // Скрываем инициал до загрузки картинки
      const txt = document.createTextNode(initial);
      avatar.appendChild(txt);
      avatar.style.color = '#fff';
    } else {
      avatar.textContent = initial;
    }
    row.appendChild(avatar);

    // Имя
    const name = document.createElement('div');
    name.className = 'lb-name';
    name.textContent = displayName;
    row.appendChild(name);

    // Счёт
    const score = document.createElement('div');
    score.className = 'lb-score';
    score.textContent = fmt(p.score || 0);
    row.appendChild(score);

    list.appendChild(row);
  });
}

async function openLeaderboard() {
  FX.click();
  $('lb-list').innerHTML = '<div class="lb-empty">Загрузка рейтинга...</div>';
  $('ov-leaderboard').classList.add('show');
  
  if (window.GameSDK && GameSDK.getLeaderboard) {
    try {
      const res = await GameSDK.getLeaderboard(20);
      if (res && res.ok && res.leaderboard) {
        renderLeaderboard(res.leaderboard);
      } else {
        renderLeaderboard([]);
      }
    } catch (e) {
      console.warn('[Cosmo-Boom] Leaderboard error:', e);
      $('lb-list').innerHTML = '<div class="lb-empty">Ошибка загрузки. Попробуйте позже.</div>';
    }
  } else {
    renderLeaderboard([]);
  }
}

function pauseGame() {
  if (!G.running || G.paused) return;
  G.paused = true; stopMusic(); $('ov-pause').classList.add('show');
}
function resumeGame() {
  if (!G.paused) return;
  G.paused = false; lastTs = performance.now();
  $('ov-pause').classList.remove('show'); startMusic();
}
function exitToMenu() {
  stopLoop(); stopMusic();
  G.running = false; G.paused = false;
  hideOverlays();
  renderMenu(); showScreen('menu');
}

    return 'Я прошёл уровень ' + n + ' в «Космо-Бум» со счётом ' + fmt(G.score) + '!';

  }

}
